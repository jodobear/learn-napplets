def anonymous_namespace_probe():
    descriptor = tempfile.TemporaryFile(mode="w+b")
    row = None
    try:
        os.fchmod(descriptor.fileno(), 0o600)
        before = os.fstat(descriptor.fileno())
        identity = (before.st_dev, before.st_ino, before.st_mode, before.st_uid, before.st_nlink)
        if before.st_uid != os.getuid() or not stat.S_ISREG(before.st_mode) or stat.S_IMODE(before.st_mode) != 0o600 or before.st_nlink != 0:
            raise RuntimeError("namespace probe is not an anonymous owned mode-0600 regular file")
        payload = b"anonymous-descriptor-owned\n"
        descriptor.write(payload)
        descriptor.flush()
        os.fsync(descriptor.fileno())
        descriptor.seek(0)
        if descriptor.read() != payload:
            raise RuntimeError("namespace probe payload changed")
        after = os.fstat(descriptor.fileno())
        if (after.st_dev, after.st_ino, after.st_mode, after.st_uid, after.st_nlink) != identity:
            raise RuntimeError("namespace probe descriptor identity changed")
        row = {
            "uidOwned": True,
            "regular": True,
            "mode": "0600",
            "nlink": 0,
            "identityRetained": True,
            "payloadExact": True,
            "namespaceEntryAbsent": True,
        }
    finally:
        descriptor.close()
    if not descriptor.closed or row is None:
        raise RuntimeError("namespace probe descriptor did not close")
    row["closed"] = True
    return row
