def anonymous_namespace_probe():

        if before.st_uid != os.getuid() or not stat.S_ISREG(before.st_mode) or stat.S_IMODE(before.st_mode) != 0o600 or before.st_nlink != 0:

        descriptor.flush()
        os.fsync(descriptor.fileno())

            raise RuntimeError("namespace probe payload changed")
        after = os.fstat(descriptor.fileno())

def collect_owned_process(owner, timeout, on_chunk=None):
    process = owner["process"]
    pgid = owner["pgid"]
    selector = selectors.DefaultSelector()
    buffers = {"stdout": bytearray(), "stderr": bytearray()}
    for name in ("stdout", "stderr"):
        pipe = getattr(process, name)
        if pipe is not None:
            os.set_blocking(pipe.fileno(), False)
            selector.register(pipe, selectors.EVENT_READ, name)
    deadline = time.monotonic() + timeout
    exit_seen = None
    teardown_started = False
    try:
        while True:
            now = time.monotonic()
            if now >= deadline:
                raise TimeoutError("verification subprocess exceeded monotonic deadline")
            for key, mask in selector.select(min(0.05, deadline - now)):
                try:
                    chunk = os.read(key.fileobj.fileno(), 65536)
                except BlockingIOError:
                    continue
                if chunk:
                    buffers[key.data].extend(chunk)
                    if on_chunk is not None:
                        on_chunk(key.data, chunk)
                else:
                    selector.unregister(key.fileobj)
            returncode = process.poll()
            if returncode is None:
                continue
            if process_group_exists(pgid):
                teardown_started = True
                teardown_active_owner(signal.SIGTERM)
                raise RuntimeError("direct child exited while descendants remained in owned group")
            if not selector.get_map():
                teardown_started = True
                teardown_active_owner(signal.SIGTERM)
                return returncode, bytes(buffers["stdout"]), bytes(buffers["stderr"])
            if exit_seen is None:
                exit_seen = time.monotonic()
            elif time.monotonic() - exit_seen >= 0.25:
                teardown_started = True
                teardown_active_owner(signal.SIGTERM)
                raise RuntimeError("direct child exited while descendant retained an output pipe")
    except BaseException:
        if active_owner is owner and not teardown_started:
            teardown_active_owner(signal.SIGTERM)
        raise
    finally:
        selector.close()
        for name in ("stdout", "stderr"):
            pipe = getattr(process, name)
            if pipe is not None:
                pipe.close()
